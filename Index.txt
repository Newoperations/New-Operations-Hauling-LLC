        // FAQ Accordion Toggle
        function toggleFaq(buttonElement) {
            const content = buttonElement.nextElementSibling;
            const icon = buttonElement.querySelector('.fa-chevron-down');
            
            content.classList.toggle('hidden');
            if (content.classList.contains('hidden')) {
                icon.style.transform = 'rotate(0deg)';
            } else {
                icon.style.transform = 'rotate(180deg)';
            }
        }

        function handleQuickForm(e) {

        function handleContactSubmit(e) {

        function showToast(message) {
            const toast = document.createElement('div');
            toast.className = 'fixed top-5 right-5 z-50 bg-brand-navy text-white px-6 py-4 rounded-xl shadow-2xl border-l-4 border-brand-amber text-sm font-bold flex items-center gap-3 transition transform duration-300 translate-y-0';
            toast.innerHTML = `<i class="fa-solid fa-circle-check text-brand-amber text-lg"></i> <span>${message}</span>`;
            document.body.appendChild(toast);

            setTimeout(() => {
                toast.classList.add('opacity-0', '-translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 4000);
        }
